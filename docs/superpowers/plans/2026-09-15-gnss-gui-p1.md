# GNSS 界面 P1（可用版）实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 `gnss_bin_parser` 增加"本地网页界面 P1"：双击启动 → 选择/拖入 `.bin` → 串行队列解析 → 实时进度 → 结束统计与星座分布 → 打开输出目录；设置与队列可记忆。

**Architecture:** 单个 Python 3 进程（仅标准库）承载 HTTP 服务与**单工作线程**任务队列；用 `subprocess` 调用现有 `gnss_parser.exe` 并逐行解析其 stdout 得到实时进度与结束统计；解析完成后用流式扫描聚合 CSV 得到星座分布/行数/时间跨度；前端为一个静态页面（原生 JS/CSS），通过 REST + SSE 与后端交互；设置与队列持久化到 `gui/state.json`。

**Tech Stack:** Python 3.12 标准库（`http.server`、`subprocess`、`csv`、`json`、`queue`、`threading`、`tkinter`、`webbrowser`）、原生 JS/CSS、Windows `.bat`；测试用 `unittest`（零依赖）。**不引入任何第三方依赖、不改动 C++ 引擎。**

**设计依据：** `docs/superpowers/specs/2026-09-15-gnss-gui-design.md`

**验证基线（每个任务收尾都要跑）：**

"""假引擎：复刻 gnss_parser 的 stdout 与产出，供 GUI 测试确定性使用。

用法与真引擎一致： fake_engine.py -i <file> -o <dir> -p <prefix>
支持环境变量：FAKE_ENGINE_FAIL=1（exit 1 + 0 帧）、FAKE_ENGINE_DELAY=<秒>（每步延时）
"""
import argparse, os, sys, time

PROGRESS = [0, 25, 50, 75, 100]

def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("-i", required=True)
    ap.add_argument("-o", default="./output")
    ap.add_argument("-p", default="gnss")
    a = ap.parse_args()

    delay = float(os.environ.get("FAKE_ENGINE_DELAY", "0"))
    fail = os.environ.get("FAKE_ENGINE_FAIL") == "1"

    size_mb = os.path.getsize(a.i) / 1048576 if os.path.exists(a.i) else 0.0
    print(f"解析文件: {a.i}", flush=True)
    print(f"输出目录: {a.o}", flush=True)
    for p in PROGRESS:
        time.sleep(delay)
        print(f"  进度: {p}% ({size_mb * p / 100:.2f} MB / {size_mb:.2f} MB)", flush=True)

    os.makedirs(a.o, exist_ok=True)
    print("========== 解析完成 ==========", flush=True)
    if fail:
        print("成功解析帧数: 0", flush=True)
        print("解析失败或未找到有效帧。", flush=True)
        return 1

    rows = {"range": 4, "satvis": 0, "satvis2": 3, "bestpos": 1}
    files = []
    for kind, n in rows.items():
        if n == 0:
            continue
        path = os.path.join(a.o, f"{a.p}_{kind}.csv")
        with open(path, "w", encoding="utf-8-sig", newline="") as f:
            if kind == "range":
                f.write("GPS_Week,TOW_ms,Sat_System,Sat_PRN,GloFreq,Pseudorange_m,PsrStd_m,"

                f.write("2215,2000,0,2,0,20100000.0,0.5,-1100.0,0.01,110.0,46.0,11.0,0x00000000,GPS\n")
                f.write("2215,2000,1,6,0,21100000.0,0.5,-2100.0,0.01,130.0,43.0,11.0,0x00040000,北斗\n")
            elif kind == "satvis2":
                f.write("GPS_Week,TOW_ms,Sat_System,Sat_PRN,Elevation_deg,Azimuth_deg,Health,"
                        "GloFreq,TrueDoppler_Hz,ApparentDoppler_Hz,Sat_System_Name\n")
                f.write("2215,1000,0,1,45.0,120.0,0,0,1000.0,1000.5,GPS\n")
                f.write("2215,1000,1,5,30.0,220.0,0,0,900.0,900.5,北斗\n")
                f.write("2215,2000,0,2,50.0,130.0,0,0,1010.0,1010.5,GPS\n")
            else:
                f.write("GPS_Week,TOW_ms,Solution_Status,Position_Type,Latitude_deg,Longitude_deg,"
                        "Height_m,Undulation_m,LatStd_m,LonStd_m,HgtStd_m,SVs_Tracked,SVs_Used\n")
                f.write("2215,1000,0,16,30.5,114.35,39.6,-14.4,0.9,0.9,1.9,20,18\n")

- 构造签名 `JobQueue(state, repo_root, on_event, engine_cmd=None)`；`engine_cmd` 为空时用 `[detect_engine(...)]`
- 工作线程从 `queue.Queue` 取任务（FIFO），一次一个；`status` 按 `queued → running → done|failed|cancelled` 迁移，每次变更都 `state.save()` + `on_event({"type": "job", "job": ...})`
- 启动子进程：`subprocess.Popen(engine_cmd + ["-i", inputPath, "-o", outputDir, "-p", prefix], stdout=PIPE, stderr=STDOUT, text=True, encoding="utf-8", errors="replace", creationflags=CREATE_NO_WINDOW)`
- 逐行读 stdout：`parse_progress_line` 更新 `progress` 并 `on_event({"type":"progress", ...})`（限流：仅在整数百分比变化时发事件）；`parse_stats_line` 填充 `stats`；同时记录最后 20 行到 `job["tailLines"]` 供失败诊断

- `enqueue` 时对每个输入算 `unique_prefix(outputDir, sanitize_prefix(stem))`，并做**预检**：输入存在、非 0 字节、输出目录可写（试写临时文件），不满足直接 `failed` + message
- `cancel`：仅对 `running`/`queued` 有效；running 时 `proc.terminate()` → `wait(5)` → `kill()`；标记 `cancelled` 并保留 `stats.outputFiles`
- `retry`：把 `failed|cancelled|interrupted` 任务重置为 `queued` 重新入队（新 prefix 计算）
- `clear(keep_running=True)`：删除非运行中任务
- `stop()`：置退出标志、唤醒工作线程、等待 join（≤5s）

```python
"""生成合规的小 .bin（官方小端布局 + 官方 CRC-32），供 GUI 端到端测试。"""
import struct


def crc32_novatel(data: bytes) -> int:
    tbl = []
    for i in range(256):
        c = i
        for _ in range(8):
            c = (c >> 1) ^ 0xEDB88320 if c & 1 else c >> 1
        tbl.append(c & 0xFFFFFFFF)
    crc = 0
    for b in data:
        crc = ((crc >> 8) & 0x00FFFFFF) ^ tbl[(crc ^ b) & 0xFF]
    return crc & 0xFFFFFFFF


def frame(msg_id: int, body: bytes, week: int = 2215, tow_ms: int = 1000) -> bytes:
    h = bytearray(28)
    h[0:3] = b"\xAA\x44\x12"; h[3] = 28
    struct.pack_into("<H", h, 4, msg_id)        # message id（小端）
    h[6] = 0; h[7] = 0                          # binary / port 0
    struct.pack_into("<H", h, 8, len(body))     # body length
    h[13] = 180                                 # time status = FINESTEERING
    struct.pack_into("<H", h, 14, week)
    struct.pack_into("<I", h, 16, tow_ms)
    raw = bytes(h) + body
    return raw + struct.pack("<I", crc32_novatel(raw))


def range_obs(prn: int, system: int, signal: int, psr: float, cn0: float) -> bytes:
    """一条 44 字节 RANGE 观测；ch-tr-status = (系统号 << 16) | (信号类型 << 21)。"""
    o = bytearray(44)
    struct.pack_into("<H", o, 0, prn)
    struct.pack_into("<H", o, 2, 0)             # glofreq = GLONASS Frequency + 7
    struct.pack_into("<d", o, 4, psr)
    struct.pack_into("<f", o, 12, 0.5)
    struct.pack_into("<d", o, 16, -1000.0)      # adr (cycles)
    struct.pack_into("<f", o, 24, 0.01)
    struct.pack_into("<f", o, 28, 100.0)        # doppler
    struct.pack_into("<f", o, 32, cn0)
    struct.pack_into("<f", o, 36, 10.0)         # locktime
    struct.pack_into("<I", o, 40, ((system & 0x7) << 16) | ((signal & 0x1F) << 21))
    return bytes(o)


def range_body(obs: list[bytes]) -> bytes:
    return struct.pack("<I", len(obs)) + b"".join(obs)


def satvis2_body(system_enum: int, sats: list[dict]) -> bytes:
    """system_enum 用官方 Table 124（0=GPS,6=BeiDou）；每条 40 字节。"""
    head = struct.pack("<IIII", system_enum, 1, 1, len(sats))   # system / sat vis / almanac / #sat
    return head + b"".join(
        struct.pack("<IIdddd", s["prn"], s.get("health", 0), s["el"], s["az"], 1000.0, 1000.5)
        for s in sats)


def bestpos_body(lat: float, lon: float, hgt: float) -> bytes:
    """72 字节 BESTPOS 体（含 H+70/H+71 两个信号掩码字节）。"""
    b = bytearray(72)
    struct.pack_into("<II", b, 0, 0, 16)        # sol status=SOL_COMPUTED, pos type=SINGLE
    struct.pack_into("<ddd", b, 8, lat, lon, hgt)
    struct.pack_into("<f", b, 32, -14.4)        # undulation
    struct.pack_into("<I", b, 36, 61)           # datum = WGS84
    struct.pack_into("<fff", b, 40, 0.9, 0.9, 1.9)
    b[64] = 20; b[65] = 18                      # #SVs(tracked) / #solnSVs(used)
    b[66] = 18; b[67] = 5; b[68] = 0; b[69] = 6; b[70] = 8; b[71] = 15
    return bytes(b)


def sample_bin(path) -> None:
    """写出含 4 条 RANGE 观测 / 3 条 SATVIS2 / 1 条 BESTPOS 的小 BIN（约 1KB）。"""
    data = b"".join([
        frame(43, range_body([
            range_obs(1, 0, 0, 20000000.0, 45.0),     # GPS L1C/A
            range_obs(5, 4, 0, 21000000.0, 44.0),     # 北斗 B1I
            range_obs(2, 0, 17, 20100000.0, 46.0),    # GPS L2C(M)：旧 bug 会误判为北斗
            range_obs(6, 4, 9, 21100000.0, 43.0),     # 北斗 B2a
        ]), tow_ms=1000),
        frame(1043, satvis2_body(0, [{"prn": 1, "el": 45.0, "az": 120.0}]), tow_ms=1000),
        frame(1043, satvis2_body(6, [{"prn": 5, "el": 30.0, "az": 220.0}]), tow_ms=1000),
        frame(1043, satvis2_body(0, [{"prn": 2, "el": 50.0, "az": 130.0}]), tow_ms=2000),
        frame(42, bestpos_body(30.5416293576904, 114.351639106197, 39.5921), tow_ms=1000),
    ])
    with open(path, "wb") as f:
        f.write(data)
```

```python
import json, subprocess, tempfile, threading, time, unittest, urllib.request
from pathlib import Path
from gui.jobs import detect_engine
from gui.server import create_server
from tests.gui.fixtures import sample_bin

REPO = Path(__file__).resolve().parents[2]


def get(url):
    with urllib.request.urlopen(url, timeout=30) as r:
        return json.loads(r.read().decode("utf-8"))


def post(url, obj):
    req = urllib.request.Request(url, data=json.dumps(obj).encode("utf-8"), method="POST",
                                 headers={"Content-Type": "application/json"})
    with urllib.request.urlopen(req, timeout=30) as r:
        return json.loads(r.read().decode("utf-8"))


class GuiE2ETest(unittest.TestCase):
    """用真实引擎（build/Release/gnss_parser.exe）走完整链路；找不到引擎则 skip。"""

    def setUp(self):
        if detect_engine(REPO, "") is None:
            self.skipTest("未找到 build/Release/gnss_parser.exe（先编译再跑）")
        self.td = tempfile.TemporaryDirectory()
        self.root = Path(self.td.name)
        self.srv, self.port = create_server(REPO, port=0)      # engine_cmd=None → 真实引擎
        threading.Thread(target=self.srv.serve_forever, daemon=True).start()
        self.base = f"http://127.0.0.1:{self.port}"

    def tearDown(self):
        self.srv.shutdown(); self.srv.server_close(); self.td.cleanup()

    def _run_to_end(self, jid, timeout=180):
        deadline = time.time() + timeout
        while time.time() < deadline:
            job = [x for x in get(self.base + "/api/jobs")["jobs"] if x["id"] == jid][0]
            if job["status"] in ("done", "failed", "cancelled"):
                return job
            time.sleep(0.2)
        self.fail("任务超时未结束")

    def test_fixture_end_to_end_and_compare_with_direct_engine(self):
        binpath = self.root / "fixture.bin"; sample_bin(binpath)
        out = self.root / "out"
        jid = post(self.base + "/api/jobs",
                   {"paths": [str(binpath)], "outputDir": str(out)})["jobs"][0]["id"]
        job = self._run_to_end(jid)
        self.assertEqual(job["status"], "done", job.get("message"))
        self.assertEqual(job["stats"]["rangeRows"], 4)
        summ = get(f"{self.base}/api/results/{jid}/summary")
        self.assertEqual(summ["datasets"]["range"]["rows"], 4)
        self.assertEqual(summ["datasets"]["satvis2"]["rows"], 3)
        self.assertEqual(summ["datasets"]["bestpos"]["rows"], 1)
        self.assertEqual(set(summ["datasets"]["range"]["systems"]), {"0", "1"})   # GPS + 北斗
        self.assertEqual(len(summ["files"]), 3)
        self.assertEqual(sum(summ["datasets"][k]["rows"] for k in summ["datasets"]), 8)

        direct = self.root / "direct"
        subprocess.run([detect_engine(REPO, ""), "-i", str(binpath), "-o", str(direct), "-p", "d"],
                       check=True, capture_output=True)
        rows = sum(1 for _ in open(direct / "d_range.csv", encoding="utf-8-sig")) - 1
        self.assertEqual(rows, 4)                                  # 与界面结果一致

    def test_empty_and_garbage_files_fail_with_message(self):
        empty = self.root / "empty.bin"; empty.write_bytes(b"")
        garbage = self.root / "garbage.bin"; garbage.write_bytes(b"A" * 5000)
        jobs = post(self.base + "/api/jobs",
                    {"paths": [str(empty), str(garbage)],
                     "outputDir": str(self.root / "out2")})["jobs"]
        results = [self._run_to_end(j["id"]) for j in jobs]
        self.assertEqual([r["status"] for r in results], ["failed", "failed"])
        self.assertTrue(all(r["message"] for r in results))
```

3. 与**直接运行引擎**的结果对比：同一 fixture 直接跑 exe 得到的 CSV 行数/统计与通过界面得到的一致
4. 空文件（0 字节）→ 任务 `failed` 且 message 非空；垃圾文件 → `failed`
